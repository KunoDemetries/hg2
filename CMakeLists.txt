set(HG_TRANSLATED_SOURCE "" CACHE FILEPATH "Local hgtool-generated translated.cpp (optional diagnostic)")
set(HG_IOP_SOURCE "" CACHE FILEPATH "Local hgtool-generated iop-translated.cpp (optional diagnostic)")
option(HG_FAST_TRANSLATION_BUILDS "Use reduced optimization for large generated translations during development" ON)
# HG-DIAG-001: keep watch-capable builds available; throughput builds may opt out.
option(HG_EE_INSTRUCTION_TRACE "Compile EE instruction history for PC watches" ON)
# HG-DIAG-002: SIF event snapshots remain active in both configurations.
option(HG_IOP_INSTRUCTION_TRACE "Compile IOP instruction history for PC watches" ON)
if(HG_IOP_SOURCE)
  add_library(hg_iop_translated STATIC "${HG_IOP_SOURCE}")
  target_compile_definitions(hg_iop_translated PRIVATE HG_IOP_INSTRUCTION_TRACE=$<BOOL:${HG_IOP_INSTRUCTION_TRACE}>)
  target_link_libraries(hg_iop_translated PUBLIC hg_iop_image)
  if(HG_FAST_TRANSLATION_BUILDS AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(hg_iop_translated PRIVATE $<$<CONFIG:Release>:-O1>)
  endif()
  add_executable(hg_iop_diagnostic runtime/iop_diagnostic.cpp)
  target_link_libraries(hg_iop_diagnostic PRIVATE hg_iop_translated)
endif()
if(HG_TRANSLATED_SOURCE)
  get_filename_component(HG_TRANSLATED_DIRECTORY "${HG_TRANSLATED_SOURCE}" DIRECTORY)
  file(GLOB HG_TRANSLATED_SHARDS CONFIGURE_DEPENDS
       "${HG_TRANSLATED_DIRECTORY}/translated-shard-*.cpp")
  add_library(hg_ee_translated STATIC "${HG_TRANSLATED_SOURCE}" ${HG_TRANSLATED_SHARDS})
  target_compile_definitions(hg_ee_translated PRIVATE HG_EE_INSTRUCTION_TRACE=$<BOOL:${HG_EE_INSTRUCTION_TRACE}>)
  target_link_libraries(hg_ee_translated PUBLIC hg_image hg_kernel)
  if(HG_FAST_TRANSLATION_BUILDS AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(hg_ee_translated PRIVATE $<$<CONFIG:Release>:-O1>)
  elseif(HG_FAST_TRANSLATION_BUILDS AND MSVC)
    # The generated EE source is intentionally very large. /O1 keeps native
    # optimization enabled while materially reducing development compile cost
    # versus the Release default /O2; guest semantics are unchanged.
    target_compile_options(hg_ee_translated PRIVATE $<$<CONFIG:Release>:/O1>)
    # The small main unit holds the AOT VU programs and dispatcher. Optimize
    # their complete call sites for speed while keeping large EE shards /O1.
    set_source_files_properties("${HG_TRANSLATED_SOURCE}" PROPERTIES
      COMPILE_OPTIONS "$<$<CONFIG:Release>:/O2>")
  endif()
  add_executable(hg_diagnostic runtime/diagnostic.cpp)
  target_link_libraries(hg_diagnostic PRIVATE hg_ee_translated)
  if(MSVC)
    target_link_options(hg_diagnostic PRIVATE /STACK:16777216)
  endif()
  if(BUILD_TESTING)
    add_test(NAME game_prefix COMMAND hg_diagnostic --verify-prefix)
    set_tests_properties(game_prefix PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()

option(HG_OPENGL_HOST "Build the OpenGL host smoke test" ON)
option(HG_FETCH_GLFW "Fetch GLFW 3.4 if no installed GLFW package exists" OFF)
if(HG_OPENGL_HOST)
  find_package(OpenGL REQUIRED)
  find_package(glfw3 3.3 QUIET)
  if(NOT TARGET glfw AND HG_FETCH_GLFW)
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git
      GIT_TAG 7b6aead9fb88b3623e3b3725ebb42670cbe4c579 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(glfw)
  endif()
  if(NOT TARGET glfw)
    message(FATAL_ERROR "Install GLFW 3.3+ or enable HG_FETCH_GLFW; disable HG_OPENGL_HOST for headless tests.")
  endif()
  add_executable(hg_opengl_host runtime/opengl_host.cpp)
  target_link_libraries(hg_opengl_host PRIVATE hg_runtime glfw OpenGL::GL)
  if(BUILD_TESTING)
    add_executable(gs_gpu_tests tests/gs_gpu_tests.cpp runtime/gl_gs.cpp)
    target_include_directories(gs_gpu_tests PRIVATE runtime)
    target_link_libraries(gs_gpu_tests PRIVATE hg_runtime glfw OpenGL::GL)
    if(WIN32)
      target_link_libraries(gs_gpu_tests PRIVATE ole32 shell32 propsys)
    endif()
    # Requires a working OpenGL4.3 driver; run explicitly, not headless CTest.
  endif()
  if(TARGET hg_native_runtime)
    find_package(Threads REQUIRED)
    add_executable(hg_game runtime/game_host.cpp runtime/gl_gs.cpp)
    target_link_libraries(hg_game PRIVATE hg_native_runtime glfw OpenGL::GL Threads::Threads)
    # HG-DIAG-028: post-link freshness evidence, outside all native timing.
    add_custom_command(TARGET hg_game POST_BUILD
      COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/hg_build_provenance.py"
        --repo "${CMAKE_CURRENT_SOURCE_DIR}" --build "${CMAKE_CURRENT_BINARY_DIR}"
        --exe "$<TARGET_FILE:hg_game>" --configuration "$<CONFIG>"
      COMMENT "Recording linked game build provenance" COMMAND_EXPAND_LISTS VERBATIM)
    if(MSVC)
      target_link_options(hg_game PRIVATE /STACK:16777216)
    endif()
    if(WIN32)
      target_link_libraries(hg_game PRIVATE ole32 shell32 propsys)
      set_target_properties(hg_game PROPERTIES WIN32_EXECUTABLE TRUE)
      if(MSVC)
        target_link_options(hg_game PRIVATE /ENTRY:mainCRTStartup)
      endif()
    endif()
  endif()
  if(WIN32)
    target_link_libraries(hg_opengl_host PRIVATE ole32 shell32 propsys)
    set_target_properties(hg_opengl_host PROPERTIES WIN32_EXECUTABLE TRUE)
    if(MSVC)
      target_link_options(hg_opengl_host PRIVATE /ENTRY:mainCRTStartup)
    endif()
  endif()
endif()
