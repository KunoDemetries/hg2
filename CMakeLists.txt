set(HG_TRANSLATED_SOURCE "" CACHE FILEPATH "Local hgtool-generated translated.cpp (optional diagnostic)")
set(HG_IOP_SOURCE "" CACHE FILEPATH "Local hgtool-generated iop-translated.cpp (optional diagnostic)")
option(HG_FAST_TRANSLATION_BUILDS "Use GCC O1 for large generated translations during development" ON)
# HG-DIAG-001: keep watch-capable builds available; throughput builds may opt out.
option(HG_EE_INSTRUCTION_TRACE "Compile EE instruction history for PC watches" ON)
if(HG_IOP_SOURCE)
  add_library(hg_iop_translated STATIC "${HG_IOP_SOURCE}")
  target_link_libraries(hg_iop_translated PUBLIC hg_iop_image)
  if(HG_FAST_TRANSLATION_BUILDS AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(hg_iop_translated PRIVATE $<$<CONFIG:Release>:-O1>)
  endif()
  add_executable(hg_iop_diagnostic runtime/iop_diagnostic.cpp)
  target_link_libraries(hg_iop_diagnostic PRIVATE hg_iop_translated)
endif()
if(HG_TRANSLATED_SOURCE)
  add_library(hg_ee_translated STATIC "${HG_TRANSLATED_SOURCE}")
  target_compile_definitions(hg_ee_translated PRIVATE HG_EE_INSTRUCTION_TRACE=$<BOOL:${HG_EE_INSTRUCTION_TRACE}>)
  target_link_libraries(hg_ee_translated PUBLIC hg_image hg_kernel)
  if(HG_FAST_TRANSLATION_BUILDS AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(hg_ee_translated PRIVATE $<$<CONFIG:Release>:-O1>)
  endif()
  add_executable(hg_diagnostic runtime/diagnostic.cpp)
  target_link_libraries(hg_diagnostic PRIVATE hg_ee_translated)
  if(MSVC)
    target_link_options(hg_diagnostic PRIVATE /STACK:16777216)
  endif()
  if(BUILD_TESTING)
    add_test(NAME game_prefix COMMAND hg_diagnostic --verify-prefix)
    set_tests_properties(game_prefix PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()

option(HG_CDVD_BOOTSTRAP_TESTS "Test original CD/DVD server bind (requires twelve-module startup bundle)" OFF)
option(HG_REBOOT_TESTS "Test IOP image reboot (requires thirteen-module bundle including EESYNC)" OFF)
option(HG_LOADFILE_BOOTSTRAP_TESTS "Test original LOADFILE RPC server registration (requires LOADFILE bundle)" OFF)
option(HG_FILEIO_BOOTSTRAP_TESTS "Test original heap server bind (requires ten-module startup bundle)" OFF)
option(HG_SIF_BOOTSTRAP_TESTS "Test original handshake (requires six-module SIF/THREADMAN bundle)" OFF)
if(HG_TRANSLATED_SOURCE AND HG_IOP_SOURCE)
  add_executable(hg_system_diagnostic runtime/system_diagnostic.cpp)
  target_link_libraries(hg_system_diagnostic PRIVATE hg_iop_translated hg_ee_translated)
  if(MSVC)
    # Generated switch functions plus native diagnostic state exceed Windows'
    # default 1MB reserve as AOT coverage grows. This is host stack only.
    target_link_options(hg_system_diagnostic PRIVATE /STACK:16777216)
  endif()
  if(BUILD_TESTING AND HG_SIF_BOOTSTRAP_TESTS)
    add_test(NAME game_sif COMMAND hg_system_diagnostic --verify-sif)
    set_tests_properties(game_sif PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  if(BUILD_TESTING AND HG_FILEIO_BOOTSTRAP_TESTS)
    add_test(NAME game_fileio COMMAND hg_system_diagnostic --verify-services)
    set_tests_properties(game_fileio PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  if(BUILD_TESTING AND HG_CDVD_BOOTSTRAP_TESTS)
    add_test(NAME game_cdvd COMMAND hg_system_diagnostic --verify-cdvd)
    set_tests_properties(game_cdvd PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  if(BUILD_TESTING AND HG_LOADFILE_BOOTSTRAP_TESTS)
    add_test(NAME game_loadfile COMMAND hg_system_diagnostic --verify-loadfile)
    set_tests_properties(game_loadfile PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  if(BUILD_TESTING AND HG_REBOOT_TESTS)
    add_test(NAME game_reboot COMMAND hg_system_diagnostic --verify-reboot)
    set_tests_properties(game_reboot PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()

option(HG_OPENGL_HOST "Build the OpenGL host smoke test" ON)
option(HG_FETCH_GLFW "Fetch GLFW 3.4 if no installed GLFW package exists" OFF)
if(HG_OPENGL_HOST)
  find_package(OpenGL REQUIRED)
  find_package(glfw3 3.3 QUIET)
  if(NOT TARGET glfw AND HG_FETCH_GLFW)
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git
      GIT_TAG 7b6aead9fb88b3623e3b3725ebb42670cbe4c579 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(glfw)
  endif()
  if(NOT TARGET glfw)
    message(FATAL_ERROR "Install GLFW 3.3+ or enable HG_FETCH_GLFW; disable HG_OPENGL_HOST for headless tests.")
  endif()
  add_executable(hg_opengl_host runtime/opengl_host.cpp)
  target_link_libraries(hg_opengl_host PRIVATE hg_runtime glfw OpenGL::GL)
endif()
